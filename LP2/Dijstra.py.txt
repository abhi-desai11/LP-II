import heapq

def dijkstra(graph, start, n):
    dist = [float('inf')] * n
    dist[start] = 0
    min_heap = [(0, start)]

    while min_heap:
        d, u = heapq.heappop(min_heap)

        for v, w in graph[u]:
            if dist[v] > d + w:
                dist[v] = d + w
                heapq.heappush(min_heap, (dist[v], v))

    return dist

# Input
n = int(input("Enter number of vertices: "))
e = int(input("Enter number of edges: "))

graph = {i: [] for i in range(n)}

for _ in range(e):
    u, v, w = map(int, input("Enter edge (u v w): ").split())
    graph[u].append((v, w))

start = int(input("Enter starting vertex: "))
distances = dijkstra(graph, start, n)

# Output
for i in range(n):
    print(f"Shortest distance from {start} to {i} is {distances[i]}")
